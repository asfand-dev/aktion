Adding your own components

Pass ComponentSpec objects to el.registerComponents([...]). Each spec defines the source signature and a renderer that returns a DOM node. The new components show up automatically in the next getSystemPrompt() call. 

const ProductCard = {
  name: "ProductCard",
  description: "Product tile with title, price, and CTA.",
  props: [
    { name: "title", type: "string" },
    { name: "price", type: "number" },
    { name: "variant", type: "string", optional: true, enum: ["default", "featured"] },
    { name: "cta", type: "Action", optional: true },
  ],
  render: (_node, props, helpers) => {
    const card = document.createElement("div");
    card.className = "product";
    card.textContent = `${props.title} — $${props.price}`;
    return card;
  },
};

el.registerComponents([ProductCard]);
